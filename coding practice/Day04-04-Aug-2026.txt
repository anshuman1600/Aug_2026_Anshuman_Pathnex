# Create User

sudo useradd anshuman

# Create Group

sudo groupadd developers

# Create User with Home Directory and Primary Group

sudo useradd -m -g developers anshuman

# Set Password

sudo passwd anshuman

# Verify User

id anshuman

# Login as User

su - anshuman

# Verify User Exists

id anshuman

# Reset Password (if required)

sudo passwd anshuman

# Check SSH Authentication Settings

sudo grep -E "PasswordAuthentication|PubkeyAuthentication|PermitRootLogin" /etc/ssh/sshd_config

# Alternative Search

sudo grep -R "PasswordAuthentication" /etc/ssh/sshd_config /etc/ssh/sshd_config.d/

# Edit SSH Config

sudo vim /etc/ssh/sshd_config

# Search inside vim

/PasswordAuthentication

# Change

PasswordAuthentication yes

# Save and Exit

Esc
:wq

# Restart SSH Service

sudo systemctl restart sshd

# Generate SSH Key (Recommended)

ssh-keygen -t ed25519 -C "anshuman"

# If ed25519 not supported

ssh-keygen -t rsa -b 4096

# Generate Additional Key Without Overwriting Existing Key

ssh-keygen -t ed25519 -f ~/.ssh/aws-server -C "anshuman"

# Check Generated Keys

cd ~/.ssh

ls -la

# View Public Key

cat ~/.ssh/id_ed25519.pub

# OR

cat ~/.ssh/aws-server.pub

# Server Side

cd ~/.ssh

ls -la

# Open authorized_keys

vim ~/.ssh/authorized_keys

# Paste Public Key

# Verify

cat ~/.ssh/authorized_keys

# Login from Local Machine

ssh anshuman@<PUBLIC-IP>
