# Check Current Branch

git branch

# Create New Branch

git checkout -b feature-login

# Switch Branch

git checkout main

# List All Branches

git branch -a

# Add Changes

git add .

# Commit Changes

git commit -m "Added login feature"

# Merge Feature Branch

git checkout main

git merge feature-login

# Delete Merged Branch

git branch -d feature-login

# Force Delete Branch

git branch -D feature-login

# View Commit History

git log

# One Line Log

git log --oneline

# Graph View

git log --oneline --graph --all

##################################################
# Merge Conflict Demo
##################################################

git checkout -b branch1

echo "branch1 change" >> app.txt

git add .

git commit -m "Branch1 changes"

git checkout main

git checkout -b branch2

echo "branch2 change" >> app.txt

git add .

git commit -m "Branch2 changes"

git checkout main

git merge branch1

git merge branch2

# Resolve Conflict Manually

git add .

git commit -m "Resolved merge conflict"

##################################################
# Rebase
##################################################

git checkout feature-login

git rebase main

# Continue Rebase

git rebase --continue

# Abort Rebase

git rebase --abort

##################################################
# Cherry Pick
##################################################

git log --oneline

git cherry-pick <commit-id>

##################################################
# Hotfix Workflow
##################################################

git checkout main

git checkout -b hotfix-login

# Make Fix

git add .

git commit -m "Fixed login issue"

git checkout main

git merge hotfix-login

git push origin main

##################################################
# Compare Merge vs Rebase
##################################################

# Merge

git merge feature-login

# Rebase

git rebase main

##################################################
# Push Branch

git push -u origin feature-login

##################################################
# Pull Latest Changes

git pull origin main

##################################################
# Check Status

git status
