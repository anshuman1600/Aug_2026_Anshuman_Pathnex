# Day05 Coding Practice - 05-Aug-2026

# Process Management

# View current terminal processes
ps

# View all processes
ps -A

# Detailed process information
ps aux

# Full format process list
ps -ef

# Search specific process
ps -ef | grep ssh
ps -ef | grep code

# Find PID using process name
pgrep sshd
pgrep code

# Kill process using PID
kill 1234

# Kill process using name
pkill code

# Real-time process monitoring
top

--------------------------------------------------

# Grep Command

# Search text
grep "root" /etc/passwd

# Ignore case
grep -i "root" /etc/passwd

# Invert match
grep -v "root" /etc/passwd

# Show line numbers
grep -n "root" /etc/passwd

# Recursive search
grep -r "ssh" /etc
